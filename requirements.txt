# Materials Project
mp-api>=0.40

# Matminer featurization
matminer>=0.9.0

# Machine learning. scikit-learn and joblib are PINNED, not just
# lower-bounded. The shipped model (Models/binaries/LumpedRFModel.joblib)
# was pickled under scikit-learn 1.9.0. joblib's ndarray reconstruction uses
# an in-place `array.shape = ...` assignment that NumPy >= 2.5 deprecated;
# our code loads models with mmap_mode="r" which avoids that path entirely
# (see engine/predict.py), so NumPy itself does NOT need an upper bound here.
# Do not loosen the scikit-learn/joblib pins without re-verifying the model
# still loads and predicts identically.
scikit-learn==1.9.*
joblib==1.5.*
# xgboost is lower-bounded rather than pinned. The deposited gbr1 was fitted and
# pickled under 3.3.0; it has been loaded and scored under 3.4.1 and returns the
# recorded benchmark values exactly, so the artifact reads across that gap. If a
# future release stops loading it, pin to 3.3.* — the version each binary was
# written under is in Models/MANIFEST.json.
xgboost>=2.0

# Data + utilities
pandas>=2.0
numpy

# Progress + plotting
tqdm
matplotlib
# seaborn is imported by FuelLatticeParameterModelPrediction.ipynb for the
# feature-correlation bar chart, and by two of the exploratory notebooks. Nothing
# in engine/ needs it, and scripts/feature_correlations.py deliberately avoids it
# so the correlation values can be regenerated without it.
seaborn

# MCP server. Upper-bounded: server.py imports mcp.server.fastmcp.FastMCP,
# which mcp 2.x renamed to mcp.server.mcpserver.MCPServer. Without the bound a
# fresh install resolves to 2.x and server.py fails to import.
mcp>=1.2,<2
